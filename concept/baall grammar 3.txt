PROGRAM -> IMPORT STATEMENT EXPORT @Start


# imports

IMPORT -> "use" _STRING ";" IMPORT | _EPSILON


# code

STATEMENT -> ASSIGNMENT ";" STATEMENT | FUNCTION_CALL ";" STATEMENT | CONTROL_STRUCTURES STATEMENT | CONTROL_STATEMENT ";" STATEMENT | _EPSILON

ASSIGNMENT -> DECLARATION | REASSIGNMENT

DECLARATION -> TYPE ":" IDENTIFIER "=" EXPRESSION | TYPE ":" IDENTIFIER ":=" EXPRESSION | TYPE ":" IDENTIFIER

REASSIGNMENT -> IDENTIFIER SHORTHAND_OPERATOR EXPRESSION | INCREMENT_OPERATION


# algebraic expression

EXPRESSION -> "(" EXPRESSION ")" | PREFIX_OPERATOR EXPRESSION | VALUE OPTIONAL_OPERATION

OPTIONAL_OPERATION -> BINARY_OPERATOR EXPRESSION | _EPSILON

VALUE -> LITERAL | IDENTIFIER_VALUE_ACCESS | FUNCTION_CALL | INCREMENT_OPERATION


IDENTIFIER_VALUE_ACCESS -> IDENTIFIER OPTIONAL_ARRAY_INDEX

OPTIONAL_ARRAY_INDEX -> "[" expression "]" OPTIONAL_ARRAY_INDEX | _EPSILON


# array

ARRAY_LITERAL -> "[" "]" | "[" EXPRESSION ARRAY_ELEMENT_LIST "]"

ARRAY_ELEMENT_LIST -> "," EXPRESSION ARRAY_ELEMENT_LIST | _EPSILON


# struct

STRUCT_LITERAL -> STRUCT_DEFINITION | STRUCT_INITIALIZATION

# struct definition

STRUCT_DEFINITION -> "(" STRUCT_FIELD STRUCT_FIELD_LIST ")"

STRUCT_FIELD_LIST -> "," STRUCT_FIELD STRUCT_FIELD_LIST | _EPSILON

STRUCT_FIELD -> TYPE ":" IDENTIFIER


# struct initialization

STRUCT_INITIALIZATION -> "(" EXPRESSION FILLED_STRUCT_FIELD_LIST ")" | "none"

FILLED_STRUCT_FIELD_LIST -> "," EXPRESSION FILLED_STRUCT_FIELD_LIST | _EPSILON


# function definition

FUNCTION_DEFINITION -> "{" STATEMENT "}" | FUNCTION_PARAMETER FUNCTION_PARAMETER_LIST "{" STATEMENT "}"

FUNCTION_PARAMETER_LIST -> "," FUNCTION_PARAMETER FUNCTION_PARAMETER_LIST | _EPSILON

FUNCTION_PARAMETER -> TYPE ":" IDENTIFIER


# function call

FUNCTION_CALL -> IDENTIFIER_VALUE_ACCESS "(" ")" | IDENTIFIER_VALUE_ACCESS "(" EXPRESSION FUNCTION_ARGUMENT_LIST ")"

FUNCTION_ARGUMENT_LIST -> "," EXPRESSION FUNCTION_ARGUMENT_LIST | _EPSILON



# If-Else, For and While Loop

CONTROL_STRUCTURES -> FOR_LOOP | WHILE_LOOP | CONDITIONAL


CONDITIONAL -> EXPRESSION "?" "{" STATEMENT "}" ELSE_BLOCK

ELSE_BLOCK -> "\|" "{" STATEMENT "}" | "\|" CONDITIONAL


FOR_LOOP -> "for" IDENTIFIER "=" EXPRESSION ".." EXPRESSION OPTIONAL_FOR_INCREMENTOR  "{" STATEMENT "}"

OPTIONAL_FOR_INCREMENTOR -> "::" REASSIGNMENT | _EPSILON


WHILE_LOOP -> "while" EXPRESSION "{" STATEMENT "}"


# miscellaneous

CONTROL_STATEMENT -> "break" | "continue" | "return"


INCREMENT_OPERATION -> UNARY_OPERATOR IDENTIFIER | IDENTIFIER UNARY_OPERATOR

BINARY_OPERATOR -> "+" | "-" | "*" | "/" | "%" | "&" | "\|" | "^"  | "<<" | ">>" | "==" | "<" | ">" | "&&" | "\|\|"

UNARY_OPERATOR -> "++" | "--"

SHORTHAND_OPERATOR -> "=" | ":=" | "+=" | "-=" | "*=" | "/=" | "&=" | "\|=" | "^="

PREFIX_OPERATOR -> "+" | "-" | "!"

LITERAL -> PRIMITIVE_LITERAL | ARRAY_LITERAL | STRUCT_LITERAL

PRIMITIVE_LITERAL -> _STRING | _NUMBER | _BOOLEAN


# typing

TYPE -> SIMPLE_TYPE OPTIONAL_ARRAY_TYPE_DEFINITION

SIMPLE_TYPE -> "string" | "number" | "boolean" | "struct" | "function" | IDENTIFIER

OPTIONAL_ARRAY_TYPE_DEFINITION -> "[" EXPRESSION "]" OPTIONAL_ARRAY_TYPE_DEFINITION | _EPSILON

IDENTIFIER -> _IDENTIFIER


# EXPORTS

EXPORT -> "export" "{" IDENTIFIER_VALUE_ACCESS ADDITIONAL_EXPORTED_ELEMENTS "}" | _EPSILON

ADDITIONAL_EXPORTED_ELEMENTS -> "," IDENTIFIER_VALUE_ACCESS ADDITIONAL_EXPORTED_ELEMENTS | _EPSILON
